\section*{Supplementary Material}
This document provides technical details that support, but are not required to follow, the main article; the main article reports all numerical values needed to evaluate the central fidelity--utility paradox. It collects the full nomenclature, implementation formulas, calibration and hyperparameter tables, seed-robustness and per-regime summaries, and the supporting diagnostics for each part of the study. The full project codebase is not publicly released at this stage; the building emulators are provided by the open-source BOPTEST framework (Blum et al., 2021; \url{https://github.com/ibpsa/project1-boptest}) and are deployed as a Docker container as described in the run-time-environment section below. All schematic figures are conceptual diagrams and are not generated from numerical datasets unless stated otherwise.\par
\medskip\noindent\textbf{How to read this Supplement.} It is organised in three parts that mirror the main article (predictive validation; controller utility and mechanism; and transfer boundary), followed by a section of implementation-level formulas.\par
\medskip\noindent\fbox{\parbox{0.97\linewidth}{\textbf{Notation (read first).} Throughout this Supplementary Material the legacy implementation labels \texttt{v3} and \texttt{v3.5} denote \textbf{BB} (the black-box surrogate) and \textbf{GB} (the grey-box surrogate) of the main article, respectively. Figure and table labels keep the legacy \texttt{v3}/\texttt{v3.5} names for consistency with the experimental records; all main-text prose uses BB/GB.}}\par\medskip

The shaped training reward is $r_t=-(w_c\,c_t + w_e\,e_t)$, with a piecewise-linear, asymmetric comfort term
\begin{equation}
 r_{\mathrm{comfort},t} =
 \begin{cases}
 -w_u\,(T_{\ell} - T_t) + b_{\mathrm{heat}} & T_t < T_{\ell}, \\
 +\beta & T_{\ell}+\delta \le T_t \le T_h-\delta, \\
 -w_o\,(T_t - T_h) + b_{\mathrm{cool}} & T_t > T_h,
 \end{cases}
 \label{eq:comfort}
\end{equation}

\section*{BOPTEST RTE control loop and supply-temperature actuation}
The live evaluation runtime is the BOPTEST Run-Time Environment (RTE): a Dockerised
Modelica emulator with embedded baseline local-loop and supervisory controllers, a
disturbance forecaster, and a KPI calculator, all exposed through a RESTful HTTP API
(Blum et al., 2021). The test controller does not replace the plant; following the
BOPTEST formalism it \emph{overwrites} a chosen subset $u_c$ of the emulator's control
inputs while the remaining embedded loops keep running at baseline. In this study $u_c$ is
the single supply-temperature set-point of each test case, activated through the BOPTEST
overwrite pair (the point's \texttt{\_activate} flag set to $1$ and its \texttt{\_u} value
written); on \texttt{bestest\_air} this point is the supply-air-temperature set-point
\texttt{oveTSetSup}.
At every control step $\Delta t=\SI{900}{\second}$ (Option-1 synchronisation, in which the
controller defines the step and the emulator advances when an action is returned) the
controller reads measurements $y_t$ (zone temperature, CO$_2$, power) and disturbance
forecasts $\omega_t$ from the API, maps its policy output $a_t\in[-1,1]$ to the
supply-temperature command $T_{\mathrm{sup}}=18+\tfrac{a_t+1}{2}(35-18)\,^\circ$C, writes
it to the overwrite signal, and issues \texttt{POST /advance} to integrate the emulator
forward one step. The RTE then returns the next measurements and the live key performance
indicators (including the maintenance score $m_s$) computed from the closed-loop
trajectory rather than from the surrogate. For the hydronic transfer cases the same
supply-temperature command is routed through the per-test-case adapter to that case's
overwrite signal (heat-pump modulation, valve position, or supply-water set-point), so the
frozen policy interface is unchanged across test cases (Figure~\ref{fig:rte_loop}).

\section*{Detailed formulas}
The following standard or implementation-level equations were moved here to keep the main
article focused on results; they are reproduced for completeness.
\noindent\textbf{PPO clipped surrogate objective} (Schulman et al., 2017), with $\hat{A}_t$ the GAE advantage:
\begin{equation}
L_{\mathrm{CLIP}}(\theta)=\mathbb{E}_t\!\left[\min\!\big(\rho_t(\theta)\,\hat{A}_t,\;
\operatorname{clip}(\rho_t(\theta),1-\epsilon,1+\epsilon)\,\hat{A}_t\big)\right],
\qquad \rho_t(\theta)=\frac{\pi_\theta(a_t\mid o_t)}{\pi_{\theta_{\mathrm{old}}}(a_t\mid o_t)}.
\end{equation}
\noindent\textbf{Black-box (BB) multi-horizon training loss} with physical-consistency penalty:
\begin{equation}
 \mathcal{L}_{\mathrm{BB}}
 = \sum_{k\in\{1,2,4\}} \frac{1}{N}\sum_{i}\big(\That_i(t{+}k)-T_i(t{+}k)\big)^2
 + \lambda_{\mathrm{phys}}\,\mathrm{ReLU}\!\big(|\Delta\That|-\Delta T_{\max}\big).
\end{equation}
\noindent\textbf{Grey-box (GB) positive reparameterization} of the zone capacitance:
\begin{equation}
 \Czon = c_s\big(\mathrm{softplus}(\rho) + c_{\min}/c_s\big),
 \qquad \Czon \ge c_{\min} = 5\times10^4~\mathrm{J/K}.
\end{equation}
\noindent\textbf{Stage B maximum-a-posteriori identification} of $\Czon$ on the excitation subset $\mathcal{E}$ under a weak Gaussian prior:
\begin{equation}
 \widehat{C}_{\mathrm{zon}}
 = \arg\min_{C}\ \sum_{i\in\mathcal{E}}\big(\That_i(C)-T_i\big)^2
 + \frac{1}{2\sigma_p^2}\big(C-C_{\mathrm{prior}}\big)^2.
\end{equation}
\noindent\textbf{Warm-started GB heat-term scaling}: when a GB surrogate is warm-started from legacy BB weights, the inherited heat term is rescaled so that the per-step increment is dimensionally consistent at the runtime control step (the BB update lacks this scaling because its increment carries no explicit $\Delta t$):
\begin{equation}
 s = s_{\mathrm{legacy}}\,\frac{C_{\mathrm{init}}}{\Delta t_{\mathrm{legacy}}\,q_{\mathrm{scale}}}.
\end{equation}
\noindent\textbf{Transfer-gap diagnostics}: surrogate$-$live maintenance-score mismatch and mean action-gap norm:
\begin{equation}
\begin{aligned}
 \Delta m_s &= m_{s,\mathrm{surrogate}} - m_{s,\mathrm{BOPTEST}}, \\
 g_a &= \frac{1}{N}\sum_{t=1}^{N}\big\|\,a_{t,\mathrm{BOPTEST}}-a_{t,\mathrm{surrogate}}\,\big\|_{2}.
\end{aligned}
\end{equation}
